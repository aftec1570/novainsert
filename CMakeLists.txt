cmake_minimum_required(VERSION 3.25)
project(NovaInsert VERSION 1.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Supported targets (1.1.0): macOS arm64, Windows x64 (MSVC, ASIO), Linux x86_64 (ALSA). NI_PLATFORM_TAG names release assets.
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "" FORCE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "" FORCE)
  set(NI_PLATFORM_TAG macOS-arm64)
elseif(WIN32)
  if(NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64" AND NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "NovaInsert builds for Windows x64 only")
  endif()
  if(MSVC)
    add_compile_options(/permissive- /utf-8)
  endif()
  # _CRT_SECURE_NO_WARNINGS: portable fopen/strncpy stay (C4996); C4324 = padding from our own alignas (cache-line rings)
  add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
  if(MSVC)
    add_compile_options(/wd4324)
  endif()
  set(NI_PLATFORM_TAG Windows-x64)
elseif(UNIX)
  if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
    message(FATAL_ERROR "NovaInsert builds for Linux x86_64 only")
  endif()
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(NI_PLATFORM_TAG Linux-x86_64)
else()
  message(FATAL_ERROR "Unsupported platform")
endif()
# Cached so tools/package_archive.sh can read the release-asset tag from <build-dir>/CMakeCache.txt
set(NI_PLATFORM_TAG "${NI_PLATFORM_TAG}" CACHE STRING "Release asset platform tag" FORCE)

# Own code is warning-clean under -Wall -Wextra -Werror (clang/gcc) or /W4 /WX (MSVC); third-party is SYSTEM.
set(NI_WARNINGS "$<IF:$<CXX_COMPILER_ID:MSVC>,/W4;/WX,-Wall;-Wextra;-Werror>")

add_subdirectory(external/JUCE)

add_subdirectory(src)
enable_testing()
add_subdirectory(tests)
